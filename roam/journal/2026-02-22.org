    Step 4: The Script
    Open =~/bin/askcode= in your editor and paste this:
    
    #+begin_src python
    #!/usr/bin/env python3
    
    import sys
    import os
    import ollama
    from pathlib import Path
    
    def read_file_safe(filepath):
        """Read file with encoding detection."""
        try:
            with open(filepath, 'r', encoding='utf-8') as f:
                return f.read()
        except UnicodeDecodeError:
            # Try with latin-1 as fallback
            try:
                with open(filepath, 'r', encoding='latin-1') as f:
                    return f.read()
            except Exception as e:
                return f"[Error reading file: {e}]"
        except Exception as e:
            return f"[Error: {e}]"
    
    def get_file_context(files):
        """Build context string from multiple files."""
        context_parts = []
        
        for filepath in files:
            if not os.path.exists(filepath):
                print(f"Warning: {filepath} not found, skipping...")
                continue
                
            content = read_file_safe(filepath)
            context_parts.append(f"--- File: {filepath} ---\n{content}\n")
        
        return "\n".join(context_parts)
    
    def ask_about_code(question, files=None, model="qwen2.5-coder:7b"):
        """Ask a question about code files using local LLM."""
        
        # If no files specified, look for common code files in current directory
        if not files:
            cwd = Path.cwd()
            # Common code extensions
            extensions = ['.py', '.js', '.jsx', '.ts', '.tsx', '.java', '.go', '.rs', '.cpp', '.c', '.h']
            files = []
            for ext in extensions:
                files.extend([str(f) for f in cwd.glob(f'*{ext}')])
            
            if not files:
                print("No code files found in current directory.")
                print("Usage: askcode 'your question' [file1] [file2] ...")
                return
            
            print(f"Found {len(files)} code files in current directory")
        
        # Build context from files
        context = get_file_context(files)
        
        if not context.strip():
            print("No valid file content found.")
            return
        
        # Create the prompt
        system_prompt = """You are a code analysis assistant. Answer questions about the provided code clearly and concisely. 
    Focus on the specific question asked. If you reference code, quote the relevant parts."""
        
        user_prompt = f"""Here is the code:
    
    {context}
    
    Question: {question}
    
    Provide a clear, focused answer."""
    
        print(f"\n🤔 Analyzing code with {model}...\n")
        
        # Stream the response
        try:
            stream = ollama.chat(
                model=model,
                messages=[
                    {'role': 'system', 'content': system_prompt},
                    {'role': 'user', 'content': user_prompt}
                ],
                stream=True
            )
            
            for chunk in stream:
                print(chunk['message']['content'], end='', flush=True)
            
            print("\n")  # Final newline
            
        except Exception as e:
            print(f"Error: {e}")
            print("\nMake sure Ollama is running: ollama serve")
    
    def main():
        if len(sys.argv) < 2:
            print("Usage: askcode 'your question' [file1] [file2] ...")
            print("\nExamples:")
            print("  askcode 'what does this code do?'")
            print("  askcode 'explain the main function' app.py")
            print("  askcode 'what are the API endpoints?' server.js routes.js")
            sys.exit(1)
        
        question = sys.argv[1]
        files = sys.argv[2:] if len(sys.argv) > 2 else None
        
        ask_about_code(question, files)
    
    if __name__ == "__main__":
        main()
