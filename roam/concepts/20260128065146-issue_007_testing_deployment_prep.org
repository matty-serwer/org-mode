Spring Boot REST API for Work Sweet project management.

## Quick Start

```bash
# Start database
docker-compose up -d

./mvnw spring-boot:run

#+end_src
** API Endpoints
*** Authentication
#+begin_src markup
POST   /api/auth/register  - Register new user
POST   /api/auth/login     - Login and get JWT

#+end_src
*** Projects
#+begin_src markup
GET    /api/projects           - List user's projects
POST   /api/projects           - Create project
GET    /api/projects/{id}      - Get project by ID
PUT    /api/projects/{id}      - Update project
DELETE /api/projects/{id}      - Archive project

#+end_src
*** Tasks
#+begin_src markup
GET    /api/tasks              - List tasks (with filters)
POST   /api/tasks              - Create task
GET    /api/tasks/{id}         - Get task by ID
PUT    /api/tasks/{id}         - Update task
PUT    /api/tasks/{id}/position - Update position (drag & drop)
DELETE /api/tasks/{id}         - Delete task

#+end_src
*** Query Parameters for GET /api/tasks
- =projectId= - Filter by project
- =status= - Filter by status (TODO, IN_PROGRESS, DONE)
- =priority= - Filter by priority (NONE, LOW, NORMAL, HIGH, URGENT)
- =assignedToId= - Filter by assigned user

#+begin_src java
package com.worksweet.util;

import com.worksweet.model.*;
import com.worksweet.repository.ProjectRepository;
import com.worksweet.repository.TaskRepository;
import com.worksweet.repository.UserRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.boot.CommandLineRunner;
import org.springframework.context.annotation.Profile;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Component;

import java.time.LocalDate;

@Component
@Profile("dev") // Only runs in dev profile
@RequiredArgsConstructor
@Slf4j
public class DataSeeder implements CommandLineRunner {

    private final UserRepository userRepository;
    private final ProjectRepository projectRepository;
    private final TaskRepository taskRepository;
    private final PasswordEncoder passwordEncoder;

    @Override
    public void run(String... args) {
        if (userRepository.count() == 0) {
            seedData();
        }
    }

    private void seedData() {
        log.info("Seeding demo data...");

        // Create demo user
        User demo = User.builder()
                .name("Demo User")
                .email("demo@worksweet.dev")
                .password(passwordEncoder.encode("demo123"))
                .avatarColor("#8B5CF6")
                .active(true)
                .build();
        demo = userRepository.save(demo);

        // Create demo project
        Project project = Project.builder()
                .name("Work Sweet MVP")
                .description("Building the initial version")
                .color("#3B82F6")
                .owner(demo)
                .archived(false)
                .build();
        project = projectRepository.save(project);

        // Create demo tasks
        Task task1 = Task.builder()
                .title("Design database schema")
                .description("Plan out the data models")
                .status(TaskStatus.DONE)
                .priority(TaskPriority.HIGH)
                .project(project)
                .createdBy(demo)
                .position(0)
                .build();

        Task task2 = Task.builder()
                .title("Build authentication")
                .description("Implement JWT auth with Spring Security")
                .status(TaskStatus.DONE)
                .priority(TaskPriority.URGENT)
                .project(project)
                .createdBy(demo)
                .position(1)
                .build();

        Task task3 = Task.builder()
                .title("Create task CRUD endpoints")
                .description("Full CRUD with filters")
                .status(TaskStatus.IN_PROGRESS)
                .priority(TaskPriority.HIGH)
                .project(project)
                .createdBy(demo)
                .dueDate(LocalDate.now().plusDays(2))
                .position(0)
                .build();

        Task task4 = Task.builder()
                .title("Build QuickTaskInput component")
                .description("The signature frictionless task creation UI")
                .status(TaskStatus.TODO)
                .priority(TaskPriority.HIGH)
                .project(project)
                .createdBy(demo)
                .position(0)
                .build();

        taskRepository.save(task1);
        taskRepository.save(task2);
        taskRepository.save(task3);
        taskRepository.save(task4);

        log.info("Demo data seeded successfully!");
        log.info("Login with: demo@worksweet.dev / demo123");
    }
}
